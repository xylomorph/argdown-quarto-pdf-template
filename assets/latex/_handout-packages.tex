% Customize quote environment to have left border like HTML
\usepackage{etoolbox}
\mdfdefinestyle{quotestyle}{%
  linewidth=3pt,
  linecolor=black!30,
  topline=false,
  rightline=false,
  bottomline=false,
  leftline=true,
  leftmargin=10pt,              % Indents the mdframed box from the page margin
  rightmargin=10pt,
  innerleftmargin=5pt,         % Space between left line and text (negative pulls text closer)
  innerrightmargin=0pt,
  innertopmargin=0pt,            % No vertical padding inside frame (line height = content height)
  innerbottommargin=0pt,
  skipabove=0pt,                 % No extra spacing (using \vspace instead)
  skipbelow=0pt
}
% Reset quote environment's built-in indentation and apply mdframed styling
% Note: LaTeX's quote environment adds ~1.5em left/right margins by default
% We reset this to 0 so only mdframed's leftmargin controls positioning
\AtBeginEnvironment{quote}{%
  \vspace{12pt}%                 % Add space before quote (especially after headers)
  \setlength{\leftmargini}{0pt}% % Remove quote's default left indentation
  \setlength{\rightmargin}{0pt}% % Remove quote's default right indentation
  \begin{mdframed}[style=quotestyle]\color{black!60}% Grey text color
}
\AtEndEnvironment{quote}{\end{mdframed}}
